 * - reset-layout → allowed (true)
 *
 * Interactive Mode:
 * - retains existing layout semantics:
 * - initial → only if no authoritative positions exist
 * - reset-layout → allowed (true)
 * - graph-update / filter-change / focus-change / investigation / mode-switch → no automatic relayout if positions exist
 * - new-nodes → 'place-new-only'
 */
export function shouldAutoLayout(
  mode: LayoutMode,
  trigger: LayoutTrigger,
  context?: {
    hasAuthoritativePositions?: boolean;
    hasNewNodes?: boolean;
  }
): boolean | 'place-new-only' {
  if (trigger === 'reset-layout') {
    return true;
  }

  const hasAuth = Boolean(context?.hasAuthoritativePositions);

  if (mode === 'static') {
    switch (trigger) {
      case 'initial':
        return !hasAuth;
      case 'graph-update':
      case 'filter-change':
      case 'focus-change':
      case 'investigation':
      case 'mode-switch':
        return false;
      case 'new-nodes':
        return 'place-new-only';
      default:
        return false;
    }
  } else {
    // Interactive mode
    switch (trigger) {
      case 'initial':
        return !hasAuth;
      case 'graph-update':
        return !hasAuth;
      case 'filter-change':
      case 'focus-change':
      case 'investigation':
      case 'mode-switch':
        return false;
      case 'new-nodes':
        return 'place-new-only';
      default:
        return false;
    }
  }
}
